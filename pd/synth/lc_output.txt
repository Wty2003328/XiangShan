foreach lib {
    fakeram7_8192x64
    fakeram7_1024x160
    fakeram7_1024x56
    fakeram7_1024x7
    fakeram7_512x308
    fakeram7_256x24
    fakeram7_256x16
    fakeram7_128x230
    fakeram7_128x100
    fakeram7_128x50
    fakeram7_128x40
    fakeram7_64x192
    fakeram7_64x64
} {
    set lib_file ${results_dir}/${lib}/${lib}.lib
    set db_file  ${output_dir}/${lib}.db
    if {[file exists ${lib_file}]} {
        puts "RM-Info: Compiling ${lib_file} -> ${db_file}"
        read_lib ${lib_file}
        write_lib ${lib} -format db -output ${db_file}
        puts "RM-Info: Done: ${db_file}"
    } else {
        puts "RM-Warning: ${lib_file} not found, skipping."
    }
}
